# JillJill Voice — curated runtime requirements for the voice-service
# (FastAPI dispatch API in server.py + LiveKit Agents worker in agent.py).
#
# Pins match the known-working WSL venv (~/venvs/jilljill-voice, Ubuntu 26.04,
# Python 3.14, LiveKit Agents 1.8.4). For an EXACT reproducible environment use
# requirements.lock.txt (full `pip freeze` snapshot); this file is the curated
# top-level set the application actually imports. Transitive deps (livekit,
# livekit-protocol, google-*, numpy, protobuf, starlette, anyio, …) are resolved
# from these and pinned precisely in requirements.lock.txt.
#
# IMPORTANT: PyAV (av) needs native FFmpeg libs and is blocked by Windows Smart
# App Control — install and run this ONLY inside WSL/Linux, never native Windows.

# ── Realtime voice stack (LiveKit Agents 1.8.4) ──────────────────────────────
livekit-agents==1.8.4
livekit-api==1.2.1
av==19.0.1                      # PyAV — audio codecs (WSL/Linux only)

# LiveKit plugins actually imported by agent.py
livekit-plugins-silero==1.8.4               # VAD / turn detection helpers
livekit-plugins-deepgram==1.8.4             # STT (primary realtime path)
livekit-plugins-openai==1.8.4               # Groq (OpenAI-compatible) + OpenAI LLM/TTS
livekit-plugins-sarvam==1.8.4               # TTS (default) + Indic STT fallback
livekit-plugins-google==1.8.4               # Gemini realtime / Google TTS fallback
livekit-plugins-elevenlabs==1.8.4           # optional TTS provider
livekit-plugins-cartesia==1.8.4             # optional TTS provider
livekit-plugins-noise-cancellation==0.3.2   # BVCTelephony (opt-in)
